```python
class Solution:

    def canFinish(self, numCourses: int, prerequisites: List[List[int]]) -> bool:

        graph = defaultdict(list)

        indegree = [0] * numCourses

        for cur,pre in prerequisites:

            indegree[cur] += 1

            graph[pre].append(cur)

        q = deque()

        for i in range(numCourses):

            if indegree[i] == 0:

                q.append(i)

        while q:

            n = len(q)

            for i in range(n):

                c = q.popleft()

                numCourses -= 1

                for nc in graph[c]:

                    indegree[nc] -= 1

                    if indegree[nc] == 0:

                        q.append(nc)

        return numCourses == 0
```
